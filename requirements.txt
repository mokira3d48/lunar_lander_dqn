# Dueling DQN on LunarLander-v3. Tested with Python 3.10 (3.10 to 3.13
# have prebuilt wheels for every package below).
#
# torch from PyPI is the CUDA build on Linux. For a smaller CPU-only build,
# install torch first from the PyTorch index (see README.md):
#   pip install torch==2.8.0 --index-url https://download.pytorch.org/whl/cpu

torch==2.8.0
# numpy >= 2.0 is required: the replay buffer uses `np.bool`, which does
# not exist in numpy 1.24 to 1.26.
numpy>=2.0,<3
# The [box2d] extra installs the Box2D physics engine and pygame-ce, which
# LunarLander needs for its simulation and its window. Its `box2d` package
# only exists as wheels for Linux x86_64/i686, macOS and Windows; on Linux
# ARM use `make install`, which compiles box2d-py instead (see README.md).
gymnasium[box2d]==1.3.0
matplotlib>=3.8
loguru>=0.7
